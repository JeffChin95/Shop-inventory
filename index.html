/******************************************************* * Throttle & Gear — Shared Inventory backend * Data lives in a Google Sheet ("Inventory DB") created * automatically the first time this runs, so staff on * different devices always see the same data. * * Data model: * - Products = the catalog (one row per SKU/UPC): name, * price, cost, category, photo, units per box. * - Stock = quantities (one row per SKU + warehouse pair) * — this is what lets the same SKU sit in multiple * warehouses with separate quantities tracked. * - Transactions = an append-only movement log, one row * per stock-in / stock-out / sale, tagged with warehouse * and category for reporting. * - Invoices = one row per completed sale. *******************************************************/ var PRODUCTS_SHEET = 'Products'; var STOCK_SHEET = 'Stock'; var TRANSACTIONS_SHEET = 'Transactions'; var INVOICES_SHEET = 'Invoices'; var SETTINGS_SHEET = 'Settings'; var PRODUCTS_HEADERS = ['UPC','Name','Cost','Price','UnitsPerBox','LegacyQty','CreatedAt','Category','Photo']; var STOCK_HEADERS = ['UPC','Warehouse','Qty']; var TRANSACTIONS_HEADERS = ['ID','Timestamp','Type','UPC','Name','Qty','Balance','Note','Cost','Category','Warehouse']; var INVOICES_HEADERS = ['ID','Number','Date','CustomerName','CustomerPhone','ItemsJSON','Subtotal','DiscountType','DiscountValue','DiscountAmount','TaxRate','TaxAmount','Total']; var SETTINGS_HEADERS = ['ShopName','Address','Phone','Currency','TaxRate','Logo','InvoiceCounter','AccessCode']; var CATEGORIES = ['Seat Cover','Lock & Security','Boxes & Cases','Carpets','Helmet','Raincoat & Jacket', 'Mud Flaps & Splashguard','Handlebar/Grips','Protective Gear','Motorcycle Hooding and Clothing', 'Motorcycle Sticker/Film','Motorcycle Accessories']; var DEFAULT_WAREHOUSE = 'Main'; /* ---------- Web app entry point ---------- */ function doGet(e) { ensureDb_(); return ContentService.createTextOutput( 'Throttle & Gear backend is running.
